func mat(color: Color, roughness := 0.85) -> StandardMaterial3D:
	var m := StandardMaterial3D.new()
	m.albedo_color = color
	m.roughness = roughness
	return m

func mesh_box(parent: Node3D, pos: Vector3, size: Vector3, color: Color) -> MeshInstance3D:
	var item := MeshInstance3D.new()
	var shape := BoxMesh.new()
	shape.size = size
	item.mesh = shape
	item.material_override = mat(color)
	item.position = pos
	parent.add_child(item)
	return item

func ball(parent: Node3D, pos: Vector3, size: Vector3, color: Color) -> MeshInstance3D:
	var item := MeshInstance3D.new()
	var shape := SphereMesh.new()
	shape.radius = 0.5
	shape.height = 1.0
	item.mesh = shape
	item.scale = size
	item.position = pos
	item.material_override = mat(color)
	parent.add_child(item)
	return item

func bar(parent: Node3D, a: Vector3, b: Vector3, radius: float, color: Color) -> MeshInstance3D:
	var item := MeshInstance3D.new()
	var shape := CylinderMesh.new()
	shape.top_radius = radius
	shape.bottom_radius = radius
	shape.height = a.distance_to(b)
	item.mesh = shape
	item.material_override = mat(color)
	item.position = (a + b) / 2.0
	parent.add_child(item)
	item.quaternion = Quaternion(Vector3.UP, (b - a).normalized())
	return item

func position_segment(item: MeshInstance3D, a: Vector3, b: Vector3) -> void:
	var delta := b - a
	item.position = (a + b) * 0.5
	item.scale.y = maxf(delta.length(), 0.0001)
	if delta.length_squared() > 0.000001:
		item.quaternion = Quaternion(Vector3.UP, delta.normalized())

func update_pedaling(angle: float) -> void:
	var center := Vector3(0, 0.58, 0.05)
	var radius := 0.18
	var thigh := 0.55
	var shin := 0.55
	for i in range(legs.size()):
		var leg: Dictionary = legs[i]
		var a := angle + PI * i
		var target := Vector3(leg.x, center.y + radius * sin(a), center.z + radius * cos(a))
		var hip := Vector3(leg.x, 1.33, 0.32)
		var direction := target - hip
		var distance := clampf(direction.length(), 0.0001, thigh + shin - 0.0001)
		var unit := direction.normalized()
		var along := (thigh * thigh - shin * shin + distance * distance) / (2.0 * distance)
		var bend := sqrt(maxf(0.0, thigh * thigh - along * along))
		var normal := Vector3(0.0, -unit.z, unit.y).normalized()
		var knee := hip + unit * along - normal * bend
		leg.knee = knee
		position_segment(leg.upper, hip, knee)
		position_segment(leg.lower, knee, target)
		position_segment(leg.arm, Vector3(leg.x, center.y, center.z), target)
		leg.pedal.position = target
		leg.foot.position = target

		wheel.rotation.z = PI / 2

		bar(pivot, Vector3(-0.07, 0, 0), Vector3(0.07, 0, 0), 0.06, Color("f3db9a"))

		wheels.append(pivot)

	for pair in [[back, seat], [seat, pedal], [pedal, back], [pedal, neck], [neck, seat], [neck, front]]:

func hide_original_legs(node: Node) -> void:
	if node is MeshInstance3D:
		if "__Leg_" in node.name or "__Foot_" in node.name:
			node.visible = false
		# The supplied GLBs have vertex colors but no material or normal vectors.
		# Generate smooth normals on a copy, retaining the supplied geometry/colors.
		if node.mesh != null and node.mesh.get_surface_count() == 1:
			var arrays: Array = node.mesh.surface_get_arrays(0)
			if arrays[Mesh.ARRAY_NORMAL] == null or arrays[Mesh.ARRAY_NORMAL].size() == 0:
				var surface := SurfaceTool.new()
				surface.create_from(node.mesh, 0)
				surface.generate_normals()
				node.mesh = surface.commit()
		var vertex_material := StandardMaterial3D.new()
		vertex_material.vertex_color_use_as_albedo = true
		vertex_material.roughness = 0.82
		node.material_override = vertex_material
	for child in node.get_children():
		hide_original_legs(child)

func change_character(index: int) -> void:
	if characters.is_empty(): return
	selected_character = posmod(index, characters.size())
	for i in range(characters.size()):
		characters[i].visible = i == selected_character
	for leg in legs:
		for part in [leg.upper, leg.lower, leg.foot_mesh]:
			(part.material_override as StandardMaterial3D).albedo_color = CHARACTER_COLORS[selected_character]
	if is_instance_valid(selection_label):
		selection_label.text = "当前角色：%s" % CHARACTER_NAMES[selected_character]
	if is_instance_valid(hud):
		hud.text = "数字伙伴的森林骑行 · " + CHARACTER_NAMES[selected_character]

func make_star() -> Node3D:
	var node := Node3D.new()
	var mesh := ArrayMesh.new()
	var surface := SurfaceTool.new()
	surface.begin(Mesh.PRIMITIVE_TRIANGLES)
	# Five-point star with two faces, rendered as an independent visual object.
	for side in [-1, 1]:
		for i in range(10):
			var a := TAU * i / 10.0
			var b := TAU * (i + 1) / 10.0
			var r1 := 0.48 if i % 2 == 0 else 0.22
			var r2 := 0.48 if (i + 1) % 2 == 0 else 0.22
			surface.add_vertex(Vector3(0, 0, side * 0.04))
			surface.add_vertex(Vector3(sin(a) * r1, cos(a) * r1, side * 0.04))
			surface.add_vertex(Vector3(sin(b) * r2, cos(b) * r2, side * 0.04))
	surface.generate_normals()
	mesh = surface.commit()
	var star := MeshInstance3D.new()
	star.mesh = mesh
	var material := mat(Color("ffd75a"))
	material.cull_mode = BaseMaterial3D.CULL_DISABLED
	material.emission_enabled = true
	material.emission = Color("d88d1b")
	material.emission_energy_multiplier = 0.35
	star.material_override = material
	node.add_child(star)
	return node

func make_log() -> Node3D:
	var node := Node3D.new()
	bar(node, Vector3(-0.55, 0.27, 0), Vector3(0.55, 0.27, 0), 0.27, Color("98613e"))
	for x in [-0.56, 0.56]:
		ball(node, Vector3(x, 0.27, 0), Vector3(0.05, 0.4, 0.4), Color("e0b57e"))

func reset_ride_items() -> void:
	for item in ride_items:
		item.node.queue_free()
	ride_items.clear()
	# A different, deterministic pattern at every stop; stars and logs never overlap.
	var index: int = quiz.current_index
	for i in range(5):
		var kind := "log" if i == 2 else "star"
		var item_lane := ((i * 2 + index) % 3) - 1
		var obj := make_log() if kind == "log" else make_star()
		obj.position = Vector3(item_lane * 1.25, 0.0 if kind == "log" else 0.85, -6.5 - i * 5.0)
		add_child(obj)
		ride_items.append({"node": obj, "kind": kind, "resolved": false})

func resolve_item(kind: String, avoided: bool) -> void:
	if kind == "star":
		stars += 1
		show_message("收集到一颗小星星！")
	elif kind == "log" and not avoided:
		hits += 1
		stars = maxi(0, stars - 1)
		show_message("碰到木头啦，跳起来可以避开")
	elif kind == "log":
		show_message("跳得真好！")
	refresh_hud()

func start_jump() -> void:
	if phase == "ride" and jump_height <= 0.001 and jump_velocity == 0.0:
		jump_velocity = 5.8

func rotate_camera(yaw: float, pitch: float) -> void:
	camera_yaw += yaw
	camera_pitch = clampf(camera_pitch + pitch, 0.14, 1.05)
	update_camera()

func update_camera() -> void:
	if camera == null or rider == null: return
	var focus := Vector3(rider.position.x * 0.35, 1.05 + jump_height * 0.3, 0)
	var horizontal := cos(camera_pitch) * camera_distance
	camera.position = focus + Vector3(sin(camera_yaw) * horizontal, sin(camera_pitch) * camera_distance, cos(camera_yaw) * horizontal)
	camera.look_at(focus, Vector3.UP)

func _unhandled_key_input(event: InputEvent) -> void:
	if phase != "ride" or not event is InputEventKey or not event.pressed or event.echo: return
	if event.keycode == KEY_LEFT or event.keycode == KEY_A:
		lane = maxi(-1, lane - 1)
	elif event.keycode == KEY_RIGHT or event.keycode == KEY_D:
		lane = mini(1, lane + 1)
	elif event.keycode == KEY_SPACE or event.keycode == KEY_UP or event.keycode == KEY_W:
		start_jump()
	elif event.keycode == KEY_C:
		change_character(selected_character + 1)
	elif event.keycode == KEY_Q:
		rotate_camera(-0.35, 0.0)
	elif event.keycode == KEY_E:
		rotate_camera(0.35, 0.0)
	elif event.keycode == KEY_R:
		camera_yaw = 0.52
		camera_pitch = 0.35
		camera_distance = 8.7
		update_camera()

func _unhandled_input(event: InputEvent) -> void:
	if event is InputEventMouseButton:
		if event.button_index == MOUSE_BUTTON_LEFT or event.button_index == MOUSE_BUTTON_RIGHT:
			dragging_camera = event.pressed
		elif event.pressed and event.button_index == MOUSE_BUTTON_WHEEL_UP:
			camera_distance = maxf(4.6, camera_distance - 0.5)
			update_camera()
		elif event.pressed and event.button_index == MOUSE_BUTTON_WHEEL_DOWN:
			camera_distance = minf(13.0, camera_distance + 0.5)
			update_camera()
	elif event is InputEventMouseMotion and dragging_camera:
		rotate_camera(-event.relative.x * 0.008, event.relative.y * 0.006)

func show_message(text: String) -> void:
	message.text = text
	message_time = 1.5

func _process(delta: float) -> void:
	if phase != "ride": return
	var dt := minf(delta, 0.05)
	travel += dt * 5.0
	if message_time > 0:
		message_time -= dt
		if message_time <= 0: message.text = ""
	if jump_height > 0.0 or jump_velocity > 0.0:
		jump_velocity -= 14.0 * dt
		jump_height = maxf(0.0, jump_height + jump_velocity * dt)
		if jump_height <= 0.0: jump_velocity = 0.0
	rider.position.x = lerpf(rider.position.x, float(lane) * 1.25, 1.0 - exp(-6.0 * dt))
	rider.position.y = jump_height + sin(travel * 3.0) * 0.025
	rider.rotation.z = (float(lane) * 1.25 - rider.position.x) * -0.07
	crank += dt * 5.0
	update_pedaling(crank)
	for wheel in wheels:
		wheel.rotation.x += dt * (5.0 / 0.43)
	for obj in scenery:
		obj.position.z += dt * 5.0
		if obj.position.z > 12.0:
			obj.position.z -= 91.0
	for item in ride_items:
		if item.resolved: continue
		var obj: Node3D = item.node
		obj.position.z += dt * 5.0
		if item.kind == "star":
			obj.rotation.y += dt * 2.5
			obj.position.y = 0.9 + sin(travel * 2.0) * 0.12
		if obj.position.z >= -0.1:
			item.resolved = true
			if absf(obj.position.x - rider.position.x) < 0.62:
				resolve_item(item.kind, jump_height > 0.45)
			obj.visible = false
	update_camera()
	if travel >= 32.0:
		show_question()
extends RefCounted

var questions := [
	{"prompt": "2 + 3 = ?", "options": [4, 5, 6], "answer": 5},
	{"prompt": "7 - 2 = ?", "options": [5, 4, 6], "answer": 5},
	{"prompt": "4 + 4 = ?", "options": [7, 9, 8], "answer": 8},
	{"prompt": "9 - 3 = ?", "options": [5, 6, 7], "answer": 6},
	{"prompt": "6 + 3 = ?", "options": [9, 8, 7], "answer": 9},
]
var current_index := 0
var wrong_attempts := 0
var correct_count := 0

func current_question() -> Dictionary:
	if finished(): return {}
	return questions[current_index]

func answer(value: int) -> bool:
	if finished(): return false
	if value != current_question().answer:
		wrong_attempts += 1
		return false
	correct_count += 1
	current_index += 1
	return true

func finished() -> bool:
	return current_index >= questions.size()

func reset() -> void:
	current_index = 0
	wrong_attempts = 0
	correct_count = 0
